<p>当前位置： {{ place }}</p>
<p>页码： {{ page }} / 300</p>
<template v-for="para in all.substring(page*1000-1000,page*1000).replace(/\n+/g,'$').split('$')">
    <p>{{para}}</p>
</template>

<script setup>
import {onMounted,ref} from 'vue';
import cnchar from 'cnchar';
var place=ref(""),page=ref(1),all=ref(""),arr="",arrw="",len;
var bqs = "，，，，，，，，，，，，，，，，，\n\n。。。。。。。。“”“”“”！？："
const num = 300000;
function init(){
    for(let i=1;i<900;i++){
        for(let j=1;j<Math.max((25-i)/2,1);j++)
            arrw += cnchar.strokeToWord(i);
        arr += cnchar.strokeToWord(i);
    }
    for(let i=0;i<170;i++){
        arrw += bqs;
    }
    arr += "，\n。“”！？：";
    console.log(arrw.length,arr.length);
    len = String(arr.length).length;
    put();
}
function prd(){
    page.value = Math.max(page.value-1,1);
}
function nxt(){
    page.value = Math.min(page.value+1,300);
}
function put(){
    place.value = "";
    page.value = 1;
    all.value = "";
    for(let i=0;i<num;i++){
        let x = arrw[Math.floor(Math.random()*arrw.length)];
        if(i<10 || i>num-10)place.value += String(arr.indexOf(x));
        if(i==10)place.value += "......";
        all.value += x;
    }
}
onMounted(init);
</script>
