<input type="checkbox" id='pck' v-model="pick"/>
<label for='pck'>补全括号</label>
<input type="text" id='ipt' v-model='it'/>
<input type='button' value='导入' @click='impt'>
<input type='button' value='清空' @click='bz=[]'>
<template v-for='(cond,idx) in bz'>
    <template v-if='idx == bz.length-1'>
        <p v-html='toText(cond,true,true,0,"",xx[idx])' style="font-family:monospace;font-weight: bold;" />
        <template v-if='idx != 0'>
            <template v-if='has(cond,0)'>
                <input type='button' value='视为归约到I' @click='force("I")'/>&ensp;
                <input type='button' value='视为无法归约到I' @click='force("K")'/>
            </template>
            <template v-else>
                <input type='button' value='完成' @click='finish()'/>
            </template>
        </template>
    </template>
    <p v-else v-html='toText(cond,false,true,0,"",xx[idx])' style="font-family:monospace;font-weight: bold;"/>
</template>

<script setup>
import {ref,computed,onMounted} from 'vue';
var pick=ref(0),it,bz = ref([]),xx=ref([]),zhm=[];
function force(x){
    bz.value[bz.value.length-1] = x;
}
function finish(){
    let res = (bz.value[bz.value.length-1] == 'I')
    bz.value.pop();xx.value.pop();
    let str = zhm.pop();
    let x = bz.value[bz.value.length-1];
    if(str.length == 3){
        bz.value[bz.value.length-1] = res ? bz.value[bz.value.length-1][0][1] : bz.value[bz.value.length-1][1]
    }
    else {
        for(let i=0;i<str.length-4;i++){
            if(str[i]=='l')x = x[0];
            else x = x[1];
        }
        if(str[str.length-4] == 'l')
            x[0] = res ? x[0][0][1] : x[0][1];
        else
            x[1] = res ? x[1][0][1] : x[1][1];
    }
    clearBZ();
}
function has(x,w){
    if(typeof x == 'string'){
        if( (w >= 1 && x == 'I') ||
            (w >= 2 && x == 'K') ||
            (w >= 3 && x == 'S') ||
            (w >= 3 && x == 'Ω')
        ) return true;
        return false;
    }
    return has(x[0],w+1) || has(x[1],0);
}
function toText(x,m,t,w,s,h){
    let ret = "";
    if(typeof x == 'string'){
        if(m){
            if( (w >= 1 && x == 'I') ||
                (w >= 2 && x == 'K') ||
                (w >= 3 && x == 'S') ||
                (w >= 3 && x == 'Ω')
            )   ret =  `<span style='color : #32CD32;' onclick='setBZ("${s}","${x}")' onmouseover='overBZ("${s}","${x}")' onmouseout='clearBZ()'>${x}</span>`
            else ret = x;
        }
        else ret = x;
    }
    else if(pick.value)ret =  `(${toText(x[0],m,t,w+1,s+'l',h)} ${toText(x[1],m,t,0,s+'r',h)})`;
    else if(t) ret = `${toText(x[0],m,1,w+1,s+'l',h)} ${toText(x[1],m,0,0,s+'r',h)}`;
    else ret =  `(${toText(x[0],m,1,w+1,s+'l',h)} ${toText(x[1],m,0,0,s+'r',h)})`;
    if(h[s]){
        ret = `<span style='background-color:${h[s]}'>${ret}</span>`
    }
    return ret;
}
function impt(){
    let ipt = `(${it})`;
    let str = '';
    for(let i=0;i<ipt.length;i++){
        if(ipt[i]=='(')str += '[';
        else if(ipt[i]=='Ω')str += 'Ω';//fuck the unicode
        else if(ipt[i]==')')str += ']';
        else if(ipt[i]!=' ')str += `"${ipt[i]}"`;
    }
    str = str
        .replace(/\]\[/g,'],[')
        .replace(/\]\"/g,'],"')
        .replace(/\"\"/g,'","')
        .replace(/\"\[/g,'",[');
    let obj = JSON.parse(str);
    function cb(x){
        if(typeof x == 'string')return x;
        if(x.length == 1)return cb(x[0]);
        return [cb(x.slice(0,x.length-1)),cb(x[x.length-1])];
    }
    obj = cb(obj);
    bz.value = [obj];
    xx.value = [{}];
}
onMounted(() => {
    window.setBZ = (str,t) => {
        let x = bz.value[bz.value.length-1];
        if(t == 'I'){
            if(str.length == 1)bz.value[bz.value.length-1] = bz.value[bz.value.length-1][1];
            else {
                for(let i=0;i<str.length-2;i++){
                    if(str[i]=='l')x = x[0];
                    else x = x[1];
                }
                if(str[str.length-2] == 'l')x[0] = x[0][1];
                else x[1] = x[1][1];
            }
        }
        if(t == 'K'){
            if(str.length == 2)bz.value[bz.value.length-1] = bz.value[bz.value.length-1][0][1];
            else {
                for(let i=0;i<str.length-3;i++){
                    if(str[i]=='l')x = x[0];
                    else x = x[1];
                }
                if(str[str.length-3] == 'l')x[0] = x[0][0][1];
                else x[1] = x[1][0][1];
            }
        }
        if(t == 'S'){
            if(str.length == 3){
                bz.value[bz.value.length-1] = [
                    [bz.value[bz.value.length-1][0][0][1],bz.value[bz.value.length-1][1]],
                    [bz.value[bz.value.length-1][0][1],bz.value[bz.value.length-1][1]]
                ]
            }
            else {
                for(let i=0;i<str.length-4;i++){
                    if(str[i]=='l')x = x[0];
                    else x = x[1];
                }
                if(str[str.length-4] == 'l')
                    x[0] = [[x[0][0][0][1],x[0][1]],[x[0][0][1],x[0][1]]];
                else
                    x[1] = [[x[1][0][0][1],x[1][1]],[x[1][0][1],x[1][1]]];
            }
        }
        if(t == 'Ω'){
            xx.value[bz.value.length-1][str] = '#E0FFFF';
            zhm.push(str);
            for(let i=0;i<str.length-3;i++){
                if(str[i]=='l')x = x[0];
                else x = x[1];
            }
            console.log(x,x[0][0][1]);
            bz.value.push(x[0][0][1])
            xx.value.push({});
        }
        else clearBZ();
    };
    window.overBZ = (str,t) => {
        if(t == 'I'){
            let w = str;
            w = w.substring(0,w.length-1) + 'r';
            xx.value[bz.value.length-1][w] = '#FFA07A';
        }
        if(t == 'K'){
            let w = str;
            w = w.substring(0,w.length-1) + 'r';
            xx.value[bz.value.length-1][w] = '#FFA07A';
            w = w.substring(0,w.length-2) + 'r';
            xx.value[bz.value.length-1][w] = '#F0E68C';
        }
        if(t == 'S' || t == 'Ω'){
            let w = str;
            w = w.substring(0,w.length-1) + 'r';
            xx.value[bz.value.length-1][w] = '#FFA07A';
            w = w.substring(0,w.length-2) + 'r';
            xx.value[bz.value.length-1][w] = '#F0E68C';
            w = w.substring(0,w.length-2) + 'r';
            xx.value[bz.value.length-1][w] = '#98FB98';
        }
    };
    window.clearBZ = ()=>{
        xx.value[bz.value.length-1] = {};
    };
})
</script>
