import BigNumber from 'bignumber.js';
function sleep(ms) {
    return new Promise(resolve => setTimeout(resolve, ms));
}
function getLian(p,q){
    let arr = [];
    while(q.comparedTo(BigNumber(0))!=0){
        arr.push(p.dividedToIntegerBy(q));
        let px = p.mod(q);
        p = q;
        q = px;
    }
    if(arr[arr.length-1].comparedTo(BigNumber(1))==0) arr = arr.slice(0,arr.length-1);
    else arr[arr.length-1] = arr[arr.length-1].minus(BigNumber(1));
    return arr;
}
async function evl(){
    console.log("---");
    let x = document.getElementById('dot').value;
    if(!x.match(/^(0|([1-9][0-9]*))(\.[0-9]+)?$/)){
        document.getElementById("val").innerHTML = `<span style='color:red'>${x}不合法</span>`;
        return;
    }
    let y;
    if(x.indexOf('.')==-1)y = 0;
    else y = x.length - x.indexOf('.') - 1;
    let lqx = BigNumber(1);
    let lpx = BigNumber(0);
    let rqx = BigNumber(0);
    let rpx = BigNumber(1);
    BigNumber.config({ DECIMAL_PLACES: y + 20 })
    let lar = getLian(BigNumber(x.replace('.','')),BigNumber('1'+'0'.repeat(y)))
    let rar = getLian(BigNumber(x.replace('.','')).plus(BigNumber(1)),BigNumber('1'+'0'.repeat(y)))
    console.log(lar.map(x=>x.toNumber()),rar.map(x=>x.toNumber()));
    let uni = [];
    while(rar.length < lar.length){
        rar.push(BigNumber(0));
    }
    for(let i=0;i<lar.length;i++){
        if(lar[i].comparedTo(rar[i])==0)uni.push(lar[i]);
        else{
            if(i == lar.length-1){
                if(lar[i].isLessThan(rar[i])){
                    /* L: XXXXXL*RR
                     * R: XXXXXL*RRRRRR
                     * u: XXXXXL*
                     */
                    uni.push(lar[i]);
                }
                else{
                    /* L: XXXXXR*LLLLLL
                     * R: XXXXXR*LL
                     * u: XXXXXR*
                     */
                    uni.push(rar[i].plus(BigNumber(1)));
                }
            }
            else if(rar[i+1] == 0 && !lar[i].isLessThan(rar[i])){
                /* L: XXXXXR*LLLL*RRRR*....
                 * R: XXXXXR*LL
                 * u: XXXXXR*
                 */
                if(rar[i].plus(BigNumber(1)).comparedTo(lar[i])==0){
                    /* L: XXXXXR*LLLRRRR*....
                     * R: XXXXXR*LL
                     * u: XXXXXR*
                     */
                    if(i == lar.length-2){
                        /* L: XXXXXR*LLLRRRR*
                         * R: XXXXXR*LL
                         * u: XXXXXR*
                         */
                        uni.push(lar[i]);
                        uni.push(lar[i+1].plus(BigNumber(1)));
                    }
                    else {
                        /* L: XXXXXR*LLLRRRR*LLLL**
                         * R: XXXXXR*LL
                         * u: XXXXXR*
                         */
                        uni.push(lar[i]);
                        uni.push(lar[i+1]);
                        uni.push(BigNumber(1));
                    }
                }
                else {
                    /* L: XXXXXR*LLLL*RRRR*....
                     * R: XXXXXR*LL
                     * u: XXXXXR*
                     */
                    uni.push(lar[i].plus(BigNumber(1)));
                }
            }
            else {
                console.log("into nor")
                if(lar[i].isLessThan(rar[i]))uni.push(lar[i]);
                else uni.push(rar[i]);
            }
            break;
        }
    }
    console.log(uni.map(x=>x.toNumber()));
    for(let i=0;i<uni.length;i++){
        let rpxt = rpx;
        let rqxt = rqx;
        rpx = lpx.plus(rpx.times(uni[i]));
        rqx = lqx.plus(rqx.times(uni[i]));
        lpx = rpxt;
        lqx = rqxt;
    }
    let mp = lpx.plus(rpx);
    let mq = lqx.plus(rqx);
    let ans = mp.dividedBy(mq).toString();
    ans = ans.substring(x.length,ans.length);
    document.getElementById("val").innerHTML = 
        `${mp.toString()}/${mq.toString()}≈<span style='color:red'>${x}</span>${ans}`;
}
