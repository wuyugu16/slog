    问题 {{ qnumber }}：{{ qtext }}
    <template v-if='qtype'>
        <input type='button' value='是' @click="handleConfirm(1)">
        <input type='button' value='否' @click="handleConfirm(0)">
    </template>
    <template v-else>
        <input type='text' v-model="message">
        <input type='button' value='确认' @click="handleConfirm(message)">
    </template>

<script setup>
import BigNumber from 'bignumber.js';
import {ref,onMounted} from 'vue';
var qnumber = ref(-1), qtext = ref(""), qtype = ref(false);
var message = ref(''), func = ref(''), asking = ref(true);
var resolvePromise = null
const waitForClick = () => {
  return new Promise((resolve) => {
    resolvePromise = resolve
  })
}
const handleConfirm = (x) => {
  if (resolvePromise) {
    resolvePromise(x)
    resolvePromise = null
  }
}
async function ask(txt,tp){
    qnumber.value++;
    qtext.value = txt;
    qtype.value = tp;
    let ans = await waitForClick();
    message.value = "";
    return ans;
}
function finish(text){
    func.value = `f(x)=${text}`;
    asking.value = false;
}
async function askConst(text){
    if(await ask(`${text}是有理数吗`,1)){
        let s = await ask(`${text}是正数吗`,1)
        if(await ask(`${text}是整数吗`,1)){
            let n = await ask(`${text}为`,0);
            return n;
        }
        let u = await ask(`${text}的分子是`,0)
        let d = await ask(`${text}的分母是`,0)
        return `${s?'':'-'}${u}/${d}`;
    }
}

                let cs = await ask("整式的次数为",0);
                let xs = await ask("整式的项数为",0);
                if(xs == 1){
                    let x = await askConst("整式的首项系数")
                    finish(`${x=='1'?'':(x=='-1'?'-':x)}x^${cs}`);
                }
                if(xs == 2){
                    let x1 = await askConst("整式的首项系数")
                    let x2 = await askConst("整式的第二项系数")
                    let x3 = await ask("整式的第二项次数",0)

                    }`x);*/
                }
            }
        }
    }

    }
}
onMounted(main);
