将<input type='text' id='p' @input='evl()'/>的Goodstein 数列展开到第<input type='text' id='b' @input='evl()'/>项：
<p id='val' v-html="res"></p>

<script setup>
import BigNumber from 'bignumber.js';
import {ref} from 'vue';
var res = ref("");
function log(type,col,ord,base){
    res.value += 
`<span style="color:${col}">[${type}]</span> ${ord} base:${base.toString()} 第${base.minus(BigNumber(1)).toString()}项<br/>`
}
function fin(ord,base){
    res.value += 
`<span style="color:green">[结束]</span> ${ord} base:${base.toString()} 第${base.minus(BigNumber(1)).toString()}项<br/>`
}
function tord(n){
    let s = n.toString(2);
    if(s == '0')return 0;
    if(s == '1')return [[0,1]];
    let ans = [];
    for(let i=0; i<s.length; i++){
        if(s[i]=='1')
            ans.push([tord(s.length-i-1),1])
    }
    return ans;
}
function tostr(n){
    if(typeof n == 'number' || n.comparedTo)return n.toString();
    if(n.length===0)return '0';
    if(n.length==1){
        if(n[0][0]==0)return n[0][1].toString();
        let x = tostr(n[0][0]);
        let y = n[0][1]==1?'':n[0][1];
        if(x == 1)return `ω${y}`;
        return `ω<sup>${x}</sup>${y}`
    }
    let ans = "";
    for(let i=0;i<n.length;i++){
        if(i)ans += '+';
        ans += tostr([n[i]]);
    }
    return ans;
}
function tonum(n,x){
    if(typeof n == 'number' || n.comparedTo)return BigNumber(n);
    if(n.length==1){
        if(n[0][0]==0)return BigNumber(n[0][1]);
        let z = tonum(n[0][0],x);
        if(z == 'over' || z.isGreaterThan(BigNumber(10)))return 'over'
        if(x.isGreaterThan(BigNumber(10)))return 'over';
        let ans = x.exponentiatedBy(z);
        if(ans.isGreaterThan(BigNumber(10)))return 'over'
        return ans;
    }
    let ans = BigNumber(0);
    for(let i=0;i<n.length;i++){
        let z = tonum([n[i]],x);
        if(z == 'over')return 'over';
        ans = ans.plus(z);
    }
    if(ans.isGreaterThan(BigNumber(10)))return 'over'
    return ans;
}
function ttrd(lft,mid,rgt){
    let _rgt = tostr(rgt.map((itm,idx)=>[rgt.length-idx-1,itm]).filter(x=>x[1].comparedTo(BigNumber(0))));
    let _lft = lft!=0?lft:'';
    let _lftp = lft!=0?'+':'';
    let _mid = mid?(mid=='β'?mid:tostr([[mid,1]])):'';
    let mrp = _rgt!=0 && mid ? '+' : '';
    if(lft==0 && !mid && _rgt==0)return '0';
    return `<span style='color:red'>${_lft}</span>${_lftp}<span style='color:blue'>${_mid}</span>${mrp}${_rgt!=0?_rgt:''}`;
}
function evl(){
    let p = document.getElementById('p').value;
    let b = document.getElementById('b').value;
    res.value = "";
    if(!p.match(/^([1-9][0-9]*)|0$/) || !b.match(/^[1-9][0-9]*$/)){
        res.value = "输入不合法!";
        return;
    }
    let pp = BigNumber(p);
    let bb = BigNumber(b);
    if(pp.comparedTo(BigNumber(0))==0){
        log("归零",'red','0',BigNumber(2));
        fin('0',bb);
        return;
    }
    let pbt = pp.toString(2);
    let lft=[],mid,rgt=[BigNumber(+pbt[pbt.length-2]),BigNumber(+pbt[pbt.length-1])];
    for(let i=pbt.length-3; i>=0; i--){
        if(pbt[i]=='1' && mid){
            lft = [[tord(pbt.length-i-1),1]].concat(lft)
        }
        if(pbt[i]=='1' && !mid) mid = tord(pbt.length-i-1);
    }
    lft = tostr(lft);
    log("初始化",'purple',ttrd(lft,mid,rgt),BigNumber(2));
    if(lft.length > 50){
        log("定义",'gold',`let α=<span style='color:red'>${lft}</span>`,BigNumber(2));
        lft = 'α';
    }
    let base = BigNumber(2);
    bb = bb.minus(BigNumber(1));
    while(1){
        if(rgt.filter(x=>x.comparedTo(BigNumber(0))).length == 0){
            if(mid == 'β')console.error('mid二次展开 请更换估算')
            if(lft=='0' && !mid){
                log("归零",'red','0',base);
            }
            base = base.plus(BigNumber(1));
            bb = bb.minus(BigNumber(1));
            // 展开mid
            console.log(mid);
            let t = tonum(mid,base);
            let N = 0;
            if(t == 'over'){
                log("定义",'gold',`let β=Σ(i=s<sub>${tostr(mid)}</sub>(${base})-1 ... 10)ω<sup>ord${base}(i)</sup>`,base);
                N = 10;
                mid = "β";
            }
            else{
                if(t.isGreaterThan(base)){
                    let z = `Σ(i=${t.minus(1)} ... ${base.plus(BigNumber(1))})ω<sup>ord${base}(i)</sup>+`;
log("定义",'gold',`let γ=${t.minus(1).isGreaterThan(base)?z:''}ω<sup>ω</sup>*${base.minus(BigNumber(2))}`,base);
                    mid = [[1,1]];
                    lft = lft=='0'?'γ':`${lft}+γ`;
                    N = base;
                }
                else{
                    N = t.toNumber();
                    mid = "";
                }
            }
            console.log(N);
            for(let i=0;i<N;i++)rgt[i] = base.minus(1);
            log("M展开",'DeepPink',ttrd(lft,mid,rgt),base);
            if(bb.comparedTo(BigNumber(0))==0){
                fin(ttrd(lft,mid,rgt),base);
                break;
            }
        }
        else{
            if(rgt[rgt.length-1].comparedTo(BigNumber(0))!=0){
                // 消除常数
                let k = rgt[rgt.length-1];
                if(k.isLessThan(bb)){
                    bb = bb.minus(k);
                    base = base.plus(k)
                    rgt[rgt.length-1] = BigNumber(0);
                    log("常数消去",'blue',ttrd(lft,mid,rgt),base);
                }
                else{
                    base = base.plus(bb)
                    rgt[rgt.length-1] = rgt[rgt.length-1].minus(bb);
                    bb = BigNumber(0);
                    log("*常数消去",'orange',ttrd(lft,mid,rgt),base);
                    fin(ttrd(lft,mid,rgt),base)
                    break;
                }
            }
            else{
                // 展开
                base = base.plus(BigNumber(1));
                bb = bb.minus(BigNumber(1));
                let i = rgt.length-1;
                while(rgt[i].comparedTo(BigNumber(0))==0) i--;
                rgt[i] = rgt[i].minus(BigNumber(1));
                for(i = i+1; i<rgt.length; i++){
                    rgt[i] = base.minus(BigNumber(1));
                }
                log("L展开",'brown',ttrd(lft,mid,rgt),base)
                if(bb.comparedTo(BigNumber(0))==0){
                    fin(ttrd(lft,mid,rgt),base);
                    break;
                }
            }
        }
    }
}
</script>
