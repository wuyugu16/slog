import {onMounted} from 'vue';
onMounted(()=>{
    document.getElementById("cowbell").addEventListener('change',()=>{
        document.getElementById("show").innerHTML = 
        document.getElementById("cowbell").value;
    })
    document.getElementById("F").addEventListener('submit',(e)=>{
        let message = "";
        if(!document.getElementById("password").value.match(/[A-Z]/g)){
            message += "密码需要包含大写字母！<br/>";
        }
        if(document.getElementById("password").value != document.getElementById("password2").value){
            message += "密码和确认密码不一致！<br/>";
        }
        let rd = String(Math.floor(Math.random()*1000));
        if(document.getElementById("name").value.indexOf(rd) == -1){
            message += `密码必须包含随机的三位数字（本轮为${rd}），下轮将刷新!`;
        }
        if(!document.getElementById("zp2").value.match(/^\d{6}$/)){
            message += "银行卡密码格式错误！<br/>";
        }
        if(!document.getElementById("zp2").value.match(/^\d{18}$/)){
            message += "身份证号格式错误！<br/>";
        }
        if(Math.floor(Math.random()*100000)!=114 || Math.floor(Math.random()*100000)!=114){
            message += "身份证号和姓名不匹配！<br/>";
        }
        if(Math.floor(Math.random()*100000)!=114 || Math.floor(Math.random()*100000)!=114){
            message += "验证码错误！<br/>";
        }
        document.getElementById("message").innerHTML = message;
        e.preventDefault();
    });
});
</script>
