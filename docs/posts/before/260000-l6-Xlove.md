function getFact(n) {
    if (n <= 1) return `${n} = ${n}`;
    let factors = [];
    let temp = n;
    for (let i = 2; i * i <= temp; i++) {
        while (temp % i === 0) {
            factors.push(i);
            temp /= i;
        }
    }
    if (temp > 1) factors.push(temp);
    return `${n} = ${factors.join('*')}`;
}
function getMax(n) {
    let factors = [];
    let temp = n;
    for (let i = 2; i * i <= temp; i++) {
        while (temp % i === 0) {
            factors.push(i);
            temp /= i;
        }
    }
    if (temp > 1) factors.push(temp);
    if(factors.length == 1)return 1;
    return [3,3,3,2,2,1][String(factors[factors.length-1]).length];
}
