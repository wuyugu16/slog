<input type='button' id='get' value="生成">
<p id='text'></p>

<script setup>
import { onMounted } from 'vue';
let x = [];
function add(w,s){
  w.forEach(item=>{
    item.push(s);
    x.push(item);
  })
}
add(__MY_DATA__[0],'七上')
add(__MY_DATA__[1],'七下')
add(__MY_DATA__[2],'八上')
add(__MY_DATA__[3],'八下')
add(__MY_DATA__[4],'九上')
onMounted(()=>{
    document.getElementById("get").addEventListener('click',(e)=>{
        let tp = x[Math.floor(Math.random()*x.length)];
        document.getElementById("text").innerHTML = `${tp[3]}第${tp[0]}页 ${tp[1]} ${tp[2]}`;
    })
});
</script>
